cmake_minimum_required(VERSION 3.20)
project(locadex VERSION 0.1.0 LANGUAGES C)
option(LOCADEX_BUILD_TESTS "Build migrated tests" ON)
option(LOCADEX_STRICT_WARNINGS "Enable strict compiler warnings" OFF)
file(GLOB LOCADEX_SOURCES CONFIGURE_DEPENDS src/*.c)
add_library(locadex STATIC ${LOCADEX_SOURCES})
target_include_directories(locadex PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include/locadex/spatial)
target_compile_features(locadex PUBLIC c_std_99)
if(LOCADEX_STRICT_WARNINGS)
 if(MSVC)
  target_compile_options(locadex PRIVATE /W4 /WX)
 else()
  target_compile_options(locadex PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(CMAKE_C_COMPILER_ID MATCHES "Clang")
   target_compile_options(locadex PRIVATE -Wno-newline-eof)
  endif()
 endif()
endif()
if(LOCADEX_BUILD_TESTS)
 enable_testing()
 set(LOCADEX_TESTS
  ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_morton.c
  ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_voxel.c
  ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_api_audit.c)
 foreach(test_source IN LISTS LOCADEX_TESTS)
  get_filename_component(test_name "${test_source}" NAME_WE)
  add_executable(locadex_${test_name} ${test_source})
  target_include_directories(locadex_${test_name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include/locadex/spatial)
  target_link_libraries(locadex_${test_name} PRIVATE locadex)
  if(NOT MSVC)
   target_link_libraries(locadex_${test_name} PRIVATE m)
  endif()
  if(LOCADEX_STRICT_WARNINGS)
   if(MSVC)
    target_compile_options(locadex_${test_name} PRIVATE /W4 /WX)
   else()
    target_compile_options(locadex_${test_name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(CMAKE_C_COMPILER_ID MATCHES "Clang")
     target_compile_options(locadex_${test_name} PRIVATE -Wno-newline-eof)
    endif()
   endif()
  endif()
  add_test(NAME "${test_name}" COMMAND "locadex_${test_name}")
 endforeach()
endif()
